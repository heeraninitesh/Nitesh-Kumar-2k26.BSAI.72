#!/usr/bin/env python3
"""
main.py
Main entry point for the Library Management System.
This file imports and uses the Library class and helper functions.
"""

from library import Library
from utils import get_positive_integer, display_menu


def main():
    """Main function to run the Library Management System."""
    library = Library()

    # Pre-populate with some sample data for easier testing
    library.add_book("101", "Python Programming", "John Smith", 5)
    library.add_book("102", "Data Structures", "Jane Doe", 3)
    library.add_book("103", "Machine Learning Basics", "Alice Brown", 2)

    while True:
        display_menu()
        choice = input("Enter your choice: ").strip()

        if choice == "1":
            # Add Book
            print("\n--- ADD NEW BOOK ---")
            book_id = input("Enter Book ID: ").strip()
            if not book_id:
                print("Book ID cannot be empty!")
                continue

            title = input("Enter Book Title: ").strip()
            author = input("Enter Author Name: ").strip()
            quantity = get_positive_integer("Enter Quantity: ")

            library.add_book(book_id, title, author, quantity)

        elif choice == "2":
            # View All Books
            library.view_all_books()

        elif choice == "3":
            # Search Book
            print("\n--- SEARCH BOOK ---")
            keyword = input("Enter Book ID or Title to search: ").strip()
            if keyword:
                library.search_book(keyword)
            else:
                print("Search keyword cannot be empty!")

        elif choice == "4":
            # Issue Book
            print("\n--- ISSUE BOOK ---")
            book_id = input("Enter Book ID: ").strip()
            student_name = input("Enter Student Name: ").strip()

            if not book_id or not student_name:
                print("Book ID and Student Name are required!")
                continue

            library.issue_book(book_id, student_name)

        elif choice == "5":
            # Return Book
            print("\n--- RETURN BOOK ---")
            book_id = input("Enter Book ID: ").strip()
            student_name = input("Enter Student Name: ").strip()

            if not book_id or not student_name:
                print("Book ID and Student Name are required!")
                continue

            library.return_book(book_id, student_name)

        elif choice == "6":
            # Delete Book
            print("\n--- DELETE BOOK ---")
            book_id = input("Enter Book ID: ").strip()
            if book_id:
                library.delete_book(book_id)
            else:
                print("Book ID cannot be empty!")

        elif choice == "7":
            # View Issued Books
            library.view_issued_books()

        elif choice == "8":
            # Exit
            print("\nThank you for using Library Management System. Goodbye!")
            break

        else:
            print("Invalid choice! Please enter a number between 1 and 8.")


# Entry point of the program
if __name__ == "__main__":
    main()
