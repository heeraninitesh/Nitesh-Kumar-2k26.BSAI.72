#!/usr/bin/env python3
"""
book.py
Contains the Book class that represents a single book in the library.
"""


class Book:
    """Represents a book in the library."""

    def __init__(self, book_id, title, author, quantity):
        """Initialize a Book with ID, title, author, and quantity."""
        self.book_id = book_id
        self.title = title
        self.author = author
        self.quantity = quantity
        self.available = quantity
        # issued_to stores student names as keys, number of copies as values
        self.issued_to = {}

    def __str__(self):
        """Return a formatted string with book details."""
        return (f"ID: {self.book_id} | Title: {self.title} | Author: {self.author} | "
                f"Total: {self.quantity} | Available: {self.available}")
